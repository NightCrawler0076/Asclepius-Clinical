pulsectl

# faster-whisper uses CTranslate2 (int8 quantized) for high-performance ARM execution
faster-whisper

# PyAV audio/video decoding
# Pinned to <16 to avoid "open() got an unexpected keyword argument 'metadata_errors'"
av<16

reportlab

huggingface-hub
